DO NOT replace or remove your existing /subscriptions/{userId} rules.
Those must continue to disallow all client writes.

    match /users/{userId}/library/{bookId} {
      function isLibraryOwner() {
        return request.auth != null
          && request.auth.uid == userId
          && request.auth.token.firebase.sign_in_provider != 'anonymous';
      }

      allow get, list, delete: if isLibraryOwner();

      allow create, update: if isLibraryOwner()
        && request.resource.data.id == bookId
        && request.resource.data.keys().hasOnly([
          'id', 'title', 'author', 'subTitle', 'imageLink',
          'subscriptionRequired', 'averageRating',
          'isFinished', 'savedAt', 'finishedAt'
        ])
        && request.resource.data.keys().hasAll([
          'id', 'title', 'author', 'subTitle', 'imageLink',
          'subscriptionRequired', 'averageRating',
          'isFinished', 'savedAt', 'finishedAt'
        ])
        && request.resource.data.title is string
        && request.resource.data.author is string
        && request.resource.data.imageLink is string
        && request.resource.data.isFinished is bool;
    }
